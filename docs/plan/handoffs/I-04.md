+"""File and dataset management utilities for Sinhala Sign Language Recognition."""
+
+import json
+import os
+from pathlib import Path
+from typing import Any, Dict, Optional, Tuple
+import numpy as np
+import yaml
+
+def load_yaml_config(config_path: str = "config/config.yaml") -> Dict[str, Any]:
+    """Safely load YAML configuration file."""
+    path = Path(config_path)
+    if not path.exists():
+        raise FileNotFoundError(f"Configuration file not found: {path.resolve()}")
+    with open(path, "r", encoding="utf-8") as f:
+        return yaml.safe_load(f)
+
+def load_processed_dataset(data_dir: str = "data/processed") -> Tuple[np.ndarray, np.ndarray, Dict[str, Any]]:
+    """Load pre-processed sequences, labels, and metadata.
+    
+    Returns:
+        (X, y, metadata_dict)
+    """
+    path = Path(data_dir)
+    data_file = path / "processed_data.npy"
+    labels_file = path / "labels.npy"
+    meta_file = path / "metadata.json"
+    
+    if not data_file.exists():
+        raise FileNotFoundError(f"Missing processed data array at {data_file}")
+    if not labels_file.exists():
+        raise FileNotFoundError(f"Missing labels array at {labels_file}")
+        
+    X = np.load(data_file)
+    y = np.load(labels_file, allow_pickle=True)
+    
+    metadata = {}
+    if meta_file.exists():
+        with open(meta_file, "r", encoding="utf-8") as f:
+            metadata = json.load(f)
+            
+    return X, y, metadata
+
+def save_json(data: Any, file_path: str, indent: int = 2):
+    """Save serializable data to JSON with UTF-8 encoding."""
+    target = Path(file_path)
+    target.parent.mkdir(parents=True, exist_ok=True)
+    with open(target, "w", encoding="utf-8") as f:
+        json.dump(data, f, ensure_ascii=False, indent=indent)

+"""Visualization utilities for Sinhala Sign Language Recognition."""
+
+from pathlib import Path
+from typing import Any, Dict, List, Optional, Sequence, cast
+import matplotlib.pyplot as plt
+import numpy as np
+
+
+def plot_training_curves(
+    history: Dict[str, Any],
+    save_path: Optional[str] = "models/saved_models/training_history.png",
+    show: bool = False,
+) -> None:
+    """Plot and save training/validation accuracy and loss curves."""
+    fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(14, 5))
+    fig = cast(Any, fig)
+
+    # Accuracy
+    if "accuracy" in history:
+        ax1.plot(history["accuracy"], label="Train Accuracy", color="#2563eb", lw=2)
+    if "val_accuracy" in history:
+        ax1.plot(history["val_accuracy"], label="Val Accuracy", color="#10b981", lw=2)
+    ax1.set_title("Model Accuracy Across Epochs", fontsize=12, fontweight="bold")
+    ax1.set_xlabel("Epoch")
+    ax1.set_ylabel("Accuracy")
+    ax1.grid(True, linestyle="--", alpha=0.6)
+    ax1.legend(loc="lower right")
+
+    # Loss
+    if "loss" in history:
+        ax2.plot(history["loss"], label="Train Loss", color="#dc2626", lw=2)
+    if "val_loss" in history:
+        ax2.plot(history["val_loss"], label="Val Loss", color="#f59e0b", lw=2)
+    ax2.set_title("Loss Trajectory Across Epochs", fontsize=12, fontweight="bold")
+    ax2.set_xlabel("Epoch")
+    ax2.set_ylabel("Loss")
+    ax2.grid(True, linestyle="--", alpha=0.6)
+    ax2.legend(loc="upper right")
+
+    plt.tight_layout()
+    if save_path:
+        Path(save_path).parent.mkdir(parents=True, exist_ok=True)
+        fig.savefig(save_path, dpi=300)  # type: ignore
+    if show:
+        plt.show()
+    plt.close(fig)
+
